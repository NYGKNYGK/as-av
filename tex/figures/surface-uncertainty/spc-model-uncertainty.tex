\newcommand{\bandprofile}[4]{%
  \begin{scope}[shift={(#1,#2)}]
    \path[fill=#3!20]
      plot[smooth] coordinates {(-1,{#4}) (-.4,{.23+#4})
        (.25,{.1+#4}) (.65,{.3+#4}) (1.05,{.18+#4})}
      --plot[smooth] coordinates {(1.05,{.18-#4}) (.65,{.3-#4})
        (.25,{.1-#4}) (-.4,{.23-#4}) (-1,{-#4})}--cycle;
    \draw[#3,line width=.85pt]
      plot[smooth] coordinates {(-1,0) (-.4,.23) (.25,.1) (.65,.3) (1.05,.18)};
  \end{scope}%
}
