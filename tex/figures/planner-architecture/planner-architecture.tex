% Feature volumes, rather than separate cards for individual network layers.
\newcommand{\featurevolume}[5]{%
  \begin{scope}[shift={(#1,#2)}]
    \path[fill=#5!38,draw=#5!65,line width=.8pt]
      (0,0) rectangle (#3,#4);
    \path[fill=#5!18,draw=#5!65,line width=.8pt]
      (0,#4)--(.30,{#4+.26})--({#3+.30},{#4+.26})--(#3,#4)--cycle;
    \path[fill=#5!58,draw=#5!65,line width=.8pt]
      (#3,0)--({#3+.30},.26)--({#3+.30},{#4+.26})--(#3,#4)--cycle;
    \foreach \ratio in {.25,.50,.75}
      \draw[#5!55,line width=.4pt] (0,{#4*\ratio})--(#3,{#4*\ratio});
  \end{scope}
}
\newcommand{\networkglyph}[3]{%
  \begin{scope}[shift={(#1,#2)}]
    \foreach \layer/\count in {0/3,1/5,2/3} {
      \foreach \j in {1,...,\count} {
        \coordinate (unit-\layer-\j)
          at ({.94*\layer},{.59*(\j-(\count+1)/2)});
      }
    }
    \foreach \i in {1,...,3} \foreach \j in {1,...,5}
      \draw[#3!25,line width=.55pt] (unit-0-\i)--(unit-1-\j);
    \foreach \i in {1,...,5} \foreach \j in {1,...,3}
      \draw[#3!25,line width=.55pt] (unit-1-\i)--(unit-2-\j);
    \foreach \layer/\count in {0/3,1/5,2/3}
      \foreach \j in {1,...,\count}
        \filldraw[fill=#3!18,draw=#3!80,line width=.85pt]
          (unit-\layer-\j) circle[radius=.155];
  \end{scope}
}
\newcommand{\camera}[3]{%
  \begin{scope}[shift={(#1,#2)},rotate=#3]
    \path[draw=orange!90!black,fill=orange!18,line width=.85pt,rounded corners=.3mm]
      (-.28,-.18) rectangle (.10,.18);
    \path[draw=orange!90!black,fill=orange!8,line width=.85pt]
      (.10,-.10)--(.31,-.23)--(.31,.23)--(.10,.10)--cycle;
  \end{scope}
}
