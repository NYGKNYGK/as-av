% A few feature planes suggest hierarchical encoding without layer dimensions.
\newcommand{\featureplane}[4]{%
  \begin{scope}[shift={(#1,#2)}]
    \path[fill=blue!26,draw=blue!65,line width=.8pt]
      (0,0) rectangle (#3,#4);
    \path[fill=blue!10,draw=blue!60,line width=.8pt]
      (0,#4)--(.27,{#4+.23})--({#3+.27},{#4+.23})--(#3,#4)--cycle;
    \path[fill=blue!48,draw=blue!65,line width=.8pt]
      (#3,0)--({#3+.27},.23)--({#3+.27},{#4+.23})--(#3,#4)--cycle;
    \foreach \ratio in {.33,.67}
      \draw[blue!40,line width=.5pt] (0,{#4*\ratio})--(#3,{#4*\ratio});
  \end{scope}
}
% Schematic regression heads: seven neurons each, with no claimed layer sizes.
\newcommand{\headglyph}[2]{%
  \begin{scope}[shift={(#1,#2)}]
    \foreach \layer/\count in {0/2,1/3,2/2}
      \foreach \j in {1,...,\count}
        \coordinate (neuron-\layer-\j)
          at ({1.0*\layer},{.70*(\j-(\count+1)/2)});
    \foreach \i in {1,2} \foreach \j in {1,2,3}
      \draw[blue!30,line width=.85pt] (neuron-0-\i)--(neuron-1-\j);
    \foreach \i in {1,2,3} \foreach \j in {1,2}
      \draw[blue!30,line width=.85pt] (neuron-1-\i)--(neuron-2-\j);
    \foreach \layer/\count in {0/2,1/3,2/2}
      \foreach \j in {1,...,\count}
        \filldraw[fill=blue!16,draw=blue!80,line width=1.1pt]
          (neuron-\layer-\j) circle[radius=.18];
  \end{scope}
}
\newcommand{\camera}[3]{%
  \begin{scope}[shift={(#1,#2)},rotate=#3]
    \path[draw=teal!90!black,fill=teal!16,line width=1pt,rounded corners=.3mm]
      (-.27,-.18) rectangle (.10,.18);
    \path[draw=teal!90!black,fill=teal!6,line width=1pt]
      (.10,-.11)--(.32,-.23)--(.32,.23)--(.10,.11)--cycle;
  \end{scope}
}
